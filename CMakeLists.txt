cmake_minimum_required(VERSION 3.24)
project(dsv41 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT APPLE)
  message(FATAL_ERROR "The checkpoint verifier currently uses Apple CommonCrypto.")
endif()
add_library(dsv41_checkpoint src/model/checkpoint_atlas.cpp)
target_include_directories(dsv41_checkpoint PUBLIC include)
target_include_directories(dsv41_checkpoint SYSTEM PUBLIC third_party)
target_compile_options(dsv41_checkpoint PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
add_executable(dsv41-inspect-checkpoint tools/inspect_checkpoint/main.cpp)
target_link_libraries(dsv41-inspect-checkpoint PRIVATE dsv41_checkpoint)
add_library(dsv41_storage src/model/weights.cpp src/engram/index.cpp src/engram/store.cpp)
target_link_libraries(dsv41_storage PUBLIC dsv41_checkpoint)
target_compile_options(dsv41_storage PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(dsv41-engram-probe tools/benchmark/engram_probe.cpp)
target_link_libraries(dsv41-engram-probe PRIVATE dsv41_storage)
option(DSV41_ENABLE_MLX "Build native MLX/Metal reference components" OFF)
if(DSV41_ENABLE_MLX)
  find_package(MLX 0.32.2 EXACT CONFIG REQUIRED)
  file(READ ${CMAKE_CURRENT_SOURCE_DIR}/metal/engram/dequantize.metal DSV41_ENGRAM_KERNEL_SOURCE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS metal/engram/dequantize.metal)
  configure_file(src/engram/kernel_source.hpp.in generated/engram_kernel_source.hpp @ONLY)
  add_library(dsv41_engram_mlx src/engram/mlx.cpp)
  target_include_directories(dsv41_engram_mlx PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
  target_link_libraries(dsv41_engram_mlx PUBLIC dsv41_storage mlx)
  set_property(TARGET dsv41-engram-probe PROPERTY LINK_LIBRARIES dsv41_engram_mlx)
  target_compile_definitions(dsv41-engram-probe PRIVATE DSV41_HAVE_MLX=1)
endif()
enable_language(OBJCXX)
add_executable(dsv41-memory-probe tools/inspect_checkpoint/memory_probe.mm)
target_include_directories(dsv41-memory-probe SYSTEM PRIVATE third_party)
target_compile_options(dsv41-memory-probe PRIVATE -fobjc-arc)
target_link_libraries(dsv41-memory-probe PRIVATE "-framework Foundation" "-framework Metal")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME checkpoint_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/checkpoint/test_atlas.py $<TARGET_FILE:dsv41-inspect-checkpoint>)
  add_executable(dsv41-test-engram tests/engram/test_engram.cpp)
  target_link_libraries(dsv41-test-engram PRIVATE dsv41_storage)
  add_test(NAME engram_contract COMMAND dsv41-test-engram)
  add_test(NAME engram_runner COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/engram/test_runner.py)
endif()
