cmake_minimum_required(VERSION 3.24)
project(dsv41 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT APPLE)
  message(FATAL_ERROR "The checkpoint verifier currently uses Apple CommonCrypto.")
endif()
add_library(dsv41_checkpoint src/model/checkpoint_atlas.cpp)
target_include_directories(dsv41_checkpoint PUBLIC include)
target_include_directories(dsv41_checkpoint SYSTEM PUBLIC third_party)
target_compile_options(dsv41_checkpoint PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
add_executable(dsv41-inspect-checkpoint tools/inspect_checkpoint/main.cpp)
target_link_libraries(dsv41-inspect-checkpoint PRIVATE dsv41_checkpoint)
enable_language(OBJCXX)
add_executable(dsv41-memory-probe tools/inspect_checkpoint/memory_probe.mm)
target_include_directories(dsv41-memory-probe SYSTEM PRIVATE third_party)
target_compile_options(dsv41-memory-probe PRIVATE -fobjc-arc)
target_link_libraries(dsv41-memory-probe PRIVATE "-framework Foundation" "-framework Metal")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME checkpoint_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/checkpoint/test_atlas.py $<TARGET_FILE:dsv41-inspect-checkpoint>)
endif()
